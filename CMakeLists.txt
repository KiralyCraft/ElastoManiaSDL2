cmake_minimum_required(VERSION 3.18)
project(ElmaClassicSDL2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ELMA_SHAREWARE "Use original SARVARI resource edition instead of registered 1.0" OFF)
option(ELMA_X86_COMPAT "Use the x86 x87 compatibility arithmetic backend" ON)
option(ELMA_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(ELMA_BUILD_TESTS "Build preservation regression tests" ON)
set(ELMA_SDL2_INCLUDE_DIR "" CACHE PATH "Manual SDL2 include directory (normally leave empty)")
set(ELMA_SDL2_LIBRARY "" CACHE FILEPATH "Manual SDL2 library (normally leave empty)")
if(ELMA_SDL2_INCLUDE_DIR AND ELMA_SDL2_LIBRARY)
    add_library(SDL2::SDL2 UNKNOWN IMPORTED)
    set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${ELMA_SDL2_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${ELMA_SDL2_INCLUDE_DIR}")
else()
    find_package(SDL2 2.0.18 CONFIG QUIET)
    if(NOT TARGET SDL2::SDL2)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.18)
        add_library(SDL2::SDL2 ALIAS PkgConfig::SDL2)
    endif()
endif()
include(cmake/LegacySources.cmake)
add_library(elma-engine OBJECT ${ELMA_LEGACY_SOURCES})
target_include_directories(elma-engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(elma-sdl2 platform/main.cpp platform/runtime.cpp platform/filesystem.cpp platform/input.cpp $<TARGET_OBJECTS:elma-engine>)
target_link_libraries(elma-sdl2 PRIVATE SDL2::SDL2)
target_include_directories(elma-sdl2 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set(ELMA_TARGETS elma-engine elma-sdl2)
if(ELMA_BUILD_TESTS)
    enable_testing()
    add_executable(elma-preservation-tests tests/preservation.cpp platform/runtime.cpp platform/filesystem.cpp platform/input.cpp $<TARGET_OBJECTS:elma-engine>)
    target_link_libraries(elma-preservation-tests PRIVATE SDL2::SDL2)
    target_include_directories(elma-preservation-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    list(APPEND ELMA_TARGETS elma-preservation-tests)
    add_test(NAME preservation COMMAND elma-preservation-tests)
    add_test(NAME malformed-resource COMMAND elma-preservation-tests malformed-resource)
    set_tests_properties(preservation malformed-resource PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0;UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
endif()
foreach(target IN LISTS ELMA_TARGETS)
    target_compile_definitions(${target} PRIVATE SDL_MAIN_HANDLED)
    if(ELMA_SHAREWARE)
        target_compile_definitions(${target} PRIVATE ELMA_SHAREWARE)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /fp:strict /W3 /wd4996)
    else()
        target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off -fno-strict-aliasing -fwrapv -Wno-write-strings)
        if(ELMA_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
    endif()
endforeach()
if(ELMA_X86_COMPAT AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86|x86)$" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    foreach(target IN LISTS ELMA_TARGETS)
        target_compile_definitions(${target} PRIVATE ELMA_X87=1)
        target_compile_options(${target} PRIVATE -mfpmath=387 -fexcess-precision=fast -fno-tree-vectorize -fno-tree-slp-vectorize)
    endforeach()
    message(STATUS "x86 x87 compatibility target: 53-bit precision; original-binary parity still requires validation")
else()
    message(WARNING "EXPERIMENTAL architecture/compiler: original x86 floating-point behavior is not guaranteed")
endif()
if(ELMA_SANITIZERS AND NOT MSVC)
    target_link_options(elma-sdl2 PRIVATE -fsanitize=address,undefined)
    if(ELMA_BUILD_TESTS)
        target_link_options(elma-preservation-tests PRIVATE -fsanitize=address,undefined)
    endif()
endif()
install(TARGETS elma-sdl2 RUNTIME DESTINATION bin)
install(FILES LICENSE.md DESTINATION share/licenses/elma-classic-sdl2)
